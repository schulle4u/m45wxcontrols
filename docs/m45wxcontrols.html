314    def _set_selection(self, index, emit_events, sync_listbox=True):
315        """
316        Internal selection implementation.
317        """
318        if not self._is_valid_index(index):
319            return wx.NOT_FOUND
320
321        old_selection = self._selection
322        if index == old_selection:
323            if sync_listbox:
324                self.listbox.SetSelection(index)
325            return old_selection
326
327        if emit_events and not self._send_book_event(
328            wx.wxEVT_LISTBOOK_PAGE_CHANGING,
329            index,
330            old_selection,
331        ):
332            if sync_listbox:
333                self.listbox.SetSelection(old_selection)
334            return wx.NOT_FOUND
335
336        self._show_page(index, sync_listbox=sync_listbox)
337
338        if emit_events:
339            self._send_book_event(wx.wxEVT_LISTBOOK_PAGE_CHANGED, index, old_selection)
340
341        return old_selection
342
343    def _show_page(self, index, sync_listbox=True):
344        """
345        Show one page and hide the previously selected page.
346        """
347        old_selection = self._selection
348        self._changing_selection = True
349        self.page_area.Freeze()
350        try:
351            if self._is_valid_index(old_selection):
352                self._pages[old_selection].Hide()
353            self._pages[index].Show()
354            self._selection = index
355            if sync_listbox:
356                self.listbox.SetSelection(index)
357            self.page_area.Layout()
358        finally:
359            self.page_area.Thaw()
360            self._changing_selection = False

362    def _send_book_event(self, event_type, selection, old_selection):
363        """
364        Send a wx.BookCtrlEvent and report whether it was allowed.
365        """
366        event = wx.BookCtrlEvent(event_type, self.GetId(), selection, old_selection)
367        event.SetEventObject(self)
368        self.GetEventHandler().ProcessEvent(event)
369        return event.IsAllowed()

372        """

374        """
